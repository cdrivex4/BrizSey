<script>
  
  const latestActiveAlertUrl = "/latest-active-alert/";

  document.addEventListener('DOMContentLoaded', () => {

    /* ── BURGER TOGGLE ── */
    const burger  = document.getElementById('navBurger');
    const menu    = document.getElementById('mainNav');
    const navClose = document.getElementById('navClose');

    const openMenu = () => {
    burger.classList.add('is-active');
    menu.classList.add('is-active');
    document.body.classList.add('mobile-nav-open');
    };

    const closeMenu = () => {
    burger.classList.remove('is-active');
    menu.classList.remove('is-active');
    document.body.classList.remove('mobile-nav-open');
    };

    if (burger && menu) {
    burger.addEventListener('click', (e) => {
        e.stopImmediatePropagation();
        menu.classList.contains('is-active') ? closeMenu() : openMenu();
    });
    }

    if (navClose) {
    navClose.addEventListener('click', closeMenu);
    }

    // close on click outside
    document.addEventListener('click', (e) => {
    if (menu &&
        menu.classList.contains('is-active') &&
        !menu.contains(e.target) &&
        !burger.contains(e.target)) {
        closeMenu();
    }
    });

    // close on Escape
    document.addEventListener('keydown', (e) => {
    if (e.key === 'Escape' && menu && menu.classList.contains('is-active')) {
        closeMenu();
    }
    });

    /* ── ACTIVE ALERT FETCH ── */
    if (latestActiveAlertUrl) {
      fetch(latestActiveAlertUrl)
        .then(response => {
          if (!response.ok) throw new Error('Error fetching navbar latest active alert');
          return response.text();
        })
        .then(data => {
          if (data) {
            const $alert = $(data);
            const $alertContainer = $alert.filter("div").first();
            if ($alertContainer.length > 0) {
              const alertStyle = $alertContainer.data('alertStyle');
              if (alertStyle === 'nav_left') {
                $('.nav-left-circle-active-alert').html(data);
              } else if (alertStyle === 'nav_top') {
                $('.nav-top-scroll-active-alert').html(data);
              }
            }
          }
        })
        .catch(error => {
          console.error("NAVBAR_LATEST_ACTIVE_ALERT_ERROR:", error);
        });
    }

  });
</script>

  // plain rgb()/rgba() strings here instead of relying on those CSS functions.
  function readThemeColor(name, fallback) {
    var value = getComputedStyle(document.documentElement).getPropertyValue(name).trim();
    return value || fallback;
  }

  function hexToRgba(hex) {
    hex = hex.replace('#', '');
    if (hex.length === 3) {
      hex = hex.split('').map(function (c) { return c + c; }).join('');
    }
    var alpha = hex.length === 8 ? parseInt(hex.slice(6, 8), 16) / 255 : 1;
    return {
      r: parseInt(hex.slice(0, 2), 16),
      g: parseInt(hex.slice(2, 4), 16),
      b: parseInt(hex.slice(4, 6), 16),
      a: alpha
    };
  }

  function toColorString(hex) {
    var c = hexToRgba(hex);
    return c.a < 1
      ? 'rgba(' + c.r + ', ' + c.g + ', ' + c.b + ', ' + c.a.toFixed(2) + ')'
      : 'rgb(' + c.r + ', ' + c.g + ', ' + c.b + ')';
  }

  function mixColors(hexA, percentA, hexB) {
    var a = hexToRgba(hexA);
    var b = hexToRgba(hexB);
    var p = percentA / 100;
    var r = Math.round(a.r * p + b.r * (1 - p));
    var g = Math.round(a.g * p + b.g * (1 - p));
    var bl = Math.round(a.b * p + b.b * (1 - p));
    return 'rgb(' + r + ', ' + g + ', ' + bl + ')';
  }

  var themeBackground = readThemeColor('--background-color', '#ced9e4');

  /* Values the backend could not parse as numbers (e.g. "25-60, Max 60") come
     through as null. Keep them null so Highcharts renders a gap instead of
     Number(null) === 0, which would plot a misleading zero. */
  function toNumericSeries(values) {
    return (values || []).map(function (value) {
      return value === null || value === undefined || value === '' ? null : Number(value);

  /* ── MAP (MapLibre GL using OpenStreetMap tiles) ── */
  function renderMap() {
    const cityData = JSON.parse(document.getElementById('city-map-data').textContent);
    if (!cityData || cityData.city_y == null || cityData.city_x == null) return;

    const style = {
      version: 8,
      sources: {
        'osm-tiles': {
          type: 'raster',
          tiles: ['https://tile.openstreetmap.org/{z}/{x}/{y}.png'],
          tileSize: 256
        }
      },
      layers: [
        {
          id: 'osm-tiles-layer',
          type: 'raster',
          source: 'osm-tiles'
        }
      ]
    };

    // Initialize the map
    const map = new maplibregl.Map({
      container: 'city-map',
      style: style,
      center: [cityData.city_x, cityData.city_y],
      zoom: 7,
      scrollZoom: false,
      preserveDrawingBuffer: true
    });

    // Add navigation control (zoom buttons)
    map.addControl(new maplibregl.NavigationControl({ showCompass: false }), 'bottom-right');

    map.on('load', function () {
      // pop-up and marker
      const popupHtml = `<strong>${cityData.city_name || ''}</strong>` + 
                        (cityData.city_condition ? ` <br> <span class="tag is-info">${cityData.city_condition}</span>` : '');
      
      const popup = new maplibregl.Popup({offset: 35}).setHTML(popupHtml);

      const marker = new maplibregl.Marker({
        color: 'var(--primary-color, #0C447C)'
      })
      .setLngLat([cityData.city_x, cityData.city_y])
      .setPopup(popup)
      .addTo(map);

      marker.addClassName('wd-map-marker');
    });

  }

  // Days are switched client side against containers laid out once, so any
  // chart the new day does not draw has to be torn down explicitly. Otherwise
  // the previous day's chart stays on screen and is read as this day's.
  function destroyGraphs() {
    Object.keys(activeCharts).forEach(function (key) {
      const chart = activeCharts[key];
      if (chart && typeof chart.destroy === 'function') {
        chart.destroy();
      }
    });

    activeCharts = {};

    GRAPH_CONTAINER_IDS.forEach(function (id) {
      const container = document.getElementById(id);
      if (container) {
        container.innerHTML = '';
      }
    });
  }

  function renderGraphsForDay(dayKey) {
    destroyGraphs();

    const showNoDataMessage = function (show) {
      if (noDataMessage) {
        noDataMessage.classList.toggle('wd-hidden', !show);
      }
    };

    if (!dayKey) {
      return;
    }

    const dayData = graphPayloads[dayKey];
    if (!dayData || !dayData.time_data || !dayData.time_data.length) {
      showNoDataMessage(true);
      return;
    }

    if (dayData.has_temperature_data) {
      activeCharts.temperature = renderTemperatureGraph(dayData, dayKey);
    }

    if (dayData.has_wind_data) {
      activeCharts.wind = renderWindGraph(dayData, dayKey);
    }

    if (dayData.has_precipitation_data) {
      activeCharts.precipitation = renderPrecipitationGraph(dayData, dayKey);
    }

    if (dayData.has_air_pressure_data) {
      activeCharts.airPressure = renderAirPressureGraph(dayData, dayKey);
    }

    if (dayData.has_humidity_data) {
      activeCharts.humidity = renderHumidityGraph(dayData, dayKey);
    }

    showNoDataMessage(Object.keys(activeCharts).length === 0);
  }

    const element = document.getElementById('wd-page');
    const cityName = document.querySelector('.wd-city') ? document.querySelector('.wd-city').textContent.trim() : 'weather';
    const activeDay = document.querySelector('.wd-day-tab.active') ? document.querySelector('.wd-day-tab.active').textContent.trim() : '';
    const filename = `${cityName.replace(/\s+/g, '_')}_${activeDay.replace(/\s+/g, '_')}_wd_page.pdf`;
    const opt = {
        margin:       0.5,
        filename:     filename,
        image:        { type: 'jpeg', quality: 1 },
        html2canvas:  {
          scale: 2,
          useCORS: true,
          scrollX: 0,
          scrollY: 0
        },
        jsPDF:        { unit: 'in', format: 'a4', orientation: 'portrait' },
        pagebreak:    { mode: ['avoid-all', 'css', 'legacy'] }
    };

    // html2pdf clones the page into a capture container that's forced to the
    // PDF's printable width (~7.27in on A4 with these margins). Highcharts
    // SVGs and the forecast table keep whatever pixel width they had on the
    // live (much wider) screen, so without this they overflow that container
    // and get clipped instead of scaled. Temporarily narrow the live layout
    // to fit the printable width and reflow the charts before capturing,
    // then restore the normal responsive layout afterwards.
    function restoreLayout() {
      element.classList.remove('wd-pdf-export');
      Object.keys(activeCharts).forEach(function (key) {
        if (activeCharts[key]) activeCharts[key].reflow();
      });
    }

    element.classList.add('wd-pdf-export');
    Object.keys(activeCharts).forEach(function (key) {
      if (activeCharts[key]) activeCharts[key].reflow();
    });

    html2pdf().from(element).set(opt).save()
      .then(restoreLayout)
      .catch(function (err) {
        restoreLayout();
        console.error(err);
      });
  });

<script>
  (function () {
    var root   = document.getElementById('sl-share');
    if (!root) return;
    var btn    = document.getElementById('sl-share-btn');
    var panel  = document.getElementById('sl-share-panel');
    var input  = document.getElementById('sl-share-url');
    var copy   = document.getElementById('sl-share-copy');
    var close  = document.getElementById('sl-share-close');
    var msg    = document.getElementById('sl-share-msg');
    var endpoint = "/api/shorten/";
    var csrftoken = "NhvnA3nbPNFQU5v0AS7WAUwzYDKBDuXVk5wHpCctjq7Nr1TYKY4sbd9QWrpoQRgV";
    var fetched = false;

    function getCookie(name) {
      var m = document.cookie.match('(^|;)\\s*' + name + '\\s*=\\s*([^;]+)');
      return m ? m.pop() : '';
    }

    function showMessage(text, isError) {
      msg.textContent = text;
      msg.classList.toggle('is-error', !!isError);
      msg.hidden = !text;
    }

    function openPanel() {
      panel.hidden = false;
      if (fetched) { input.select(); return; }

      btn.classList.add('is-loading');
      showMessage('', false);

      var body = new URLSearchParams();
      body.append('url', window.location.href);

      fetch(endpoint, {
        method: 'POST',
        headers: {
          'X-CSRFToken': csrftoken || getCookie('csrftoken'),
          'X-Requested-With': 'XMLHttpRequest',
          'Content-Type': 'application/x-www-form-urlencoded',
        },
        body: body.toString(),
      })
        .then(function (r) { return r.json().then(function (d) { return { ok: r.ok, data: d }; }); })
        .then(function (res) {
          btn.classList.remove('is-loading');
          if (!res.ok) {
            showMessage(res.data.error || 'Could not create a short link.', true);
            return;
          }
          fetched = true;
          input.value = res.data.short_url;
          input.select();
          showMessage(
            res.data.created
              ? 'Short link created.'
              : 'Reusing an existing short link for this page.',
            false
          );
        })
        .catch(function () {
          btn.classList.remove('is-loading');
          showMessage('Could not create a short link.', true);
        });
    }

    function closePanel() { panel.hidden = true; }

    btn.addEventListener('click', function () {
      if (panel.hidden) { openPanel(); } else { closePanel(); }
    });
    close.addEventListener('click', closePanel);

    copy.addEventListener('click', function () {
      if (!input.value) return;
      navigator.clipboard.writeText(input.value).then(function () {
        var label = copy.textContent;
        copy.textContent = 'Copied!';
        setTimeout(function () { copy.textContent = label; }, 1500);
      });
    });

    // Close when clicking outside the widget
    document.addEventListener('click', function (e) {
      if (!panel.hidden && !root.contains(e.target)) closePanel();
    });
  })();
</script>
